//go:build temporal_integration

package temporal_test

import (
	"context"
	"github.com/MUC-WIKI/campus-agent/apps/agent-control/integration/temporalfake"
	"github.com/MUC-WIKI/campus-agent/apps/agent-control/internal/temporalruntime"
	enumspb "go.temporal.io/api/enums/v1"
	"go.temporal.io/sdk/client"
	"go.temporal.io/sdk/temporal"
	"sync"
	"testing"
	"time"
)

func TestTemporalDevServerCancelWorkflow(t *testing.T) {
	ctx, cancel := context.WithTimeout(context.Background(), 45*time.Second)
	defer cancel()
	temporalClient := packageHarness.Client
	activities := &blockingAgentActivities{Activities: temporalfake.Activities{}, started: make(chan struct{})}
	packageHarness.StartStandardWorkers(t, activities)
	runID := packageHarness.NewWorkflowID("workflow-cancel")
	input := temporalfake.WorkflowInput(runID)
	input.WorkflowID = packageHarness.NewWorkflowID(runID)
	workflowRun, err := temporalClient.ExecuteWorkflow(ctx, client.StartWorkflowOptions{
		ID:        input.WorkflowID,
		TaskQueue: temporalruntime.WorkflowTaskQueue,
	}, temporalruntime.ToolExecutionWorkflow, input)
	if err != nil {
		t.Fatalf("execute workflow: %v", err)
	}
	packageHarness.CleanupRun(t, input.WorkflowID, workflowRun.GetRunID())
	select {
	case <-activities.started:
	case <-ctx.Done():
		t.Fatalf("wait for solve activity start: %v", ctx.Err())
	}
	starter := newStarter(t, ctx)
	if err := starter.CancelToolExecutionWorkflow(ctx, temporalruntime.ToolExecutionWorkflowCancelInput{
		RequestID:     runID,
		TemporalRunID: workflowRun.GetRunID(),
		WorkflowID:    input.WorkflowID,
	}); err != nil {
		t.Fatalf("cancel workflow: %v", err)
	}
	if err := workflowRun.Get(ctx, nil); !temporal.IsCanceledError(err) {
		t.Fatalf("get canceled workflow error = %v, want canceled error", err)
	}
	history, err := packageHarness.History(ctx, input.WorkflowID, workflowRun.GetRunID())
	if err != nil {
		t.Fatalf("workflow history: %v", err)
	}
	if historyContains(history, enumspb.EVENT_TYPE_ACTIVITY_TASK_TIMED_OUT) {
		t.Fatal("cancellation depended on activity timeout")
	}
	if !historyContains(history, enumspb.EVENT_TYPE_WORKFLOW_EXECUTION_CANCELED) {
		t.Fatal("workflow history did not contain canceled event")
	}
	if !historyContainsCompletedActivity(history, temporalruntime.FinalizeExecutionCancellationActivityName) {
		t.Fatalf("workflow history missing completed activity %s", temporalruntime.FinalizeExecutionCancellationActivityName)
	}
}

type blockingAgentActivities struct {
	temporalfake.Activities
	started chan struct{}
	once    sync.Once
}

func (a *blockingAgentActivities) SolveTaskActivity(
	ctx context.Context,
	input temporalruntime.SolveTaskActivityInput,
) (temporalruntime.WorkflowOperationActivityOutput, error) {
	a.once.Do(func() { close(a.started) })
	return temporalruntime.WorkflowOperationActivityOutput{}, temporalfake.WaitForCancellation(ctx)
}
