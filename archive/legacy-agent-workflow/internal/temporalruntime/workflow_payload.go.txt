package temporalruntime

import (
	"github.com/MUC-WIKI/campus-agent/apps/agent-control/internal/execution"
	"go.temporal.io/sdk/temporal"
	"go.temporal.io/sdk/workflow"
)

func executeBoundedActivity(ctx workflow.Context, name string, input any) workflow.Future {
	if err := execution.ValidateOperationPayload(input); err != nil {
		future, result := workflow.NewFuture(ctx)
		result.Set(nil, temporal.NewNonRetryableApplicationError("Activity input exceeds the operation payload budget.", "workflow_operation_payload_too_large", nil))
		return future
	}
	return workflow.ExecuteActivity(ctx, name, input)
}
