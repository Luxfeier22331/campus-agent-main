package httpapi

import (
	"bytes"
	"encoding/json"
	"errors"
	"fmt"
	"io"
	"net/http"
	"strings"
	"time"

	"github.com/MUC-WIKI/campus-agent/apps/agent-control/internal/executionapplication"
)

const personalMemoryRecallToolName = executionapplication.PersonalMemoryRecallToolName

type personalMemoryRecallToolEnvelope struct {
	AccessContext agentToolAccessContext          `json:"access_context"`
	Payload       personalMemoryRecallToolPayload `json:"payload"`
}

type agentToolAccessContext struct {
	ExcludedResultRefs []string `json:"excluded_result_refs,omitempty"`
	ActivityAttempt    int      `json:"activity_attempt"`
	DeadlineAt         string   `json:"deadline_at"`
	RequestID          string   `json:"request_id"`
	InvocationID       string   `json:"invocation_id"`
	WorkflowStepID     string   `json:"workflow_step_id"`
	CitationStart      int      `json:"citation_start,omitempty"`
	ResultLimit        int      `json:"result_limit,omitempty"`
	UserGoal           string   `json:"user_goal,omitempty"`
}

type personalMemoryRecallToolPayload struct {
	Query string `json:"query"`
}

func (s *Server) handlePersonalMemoryRecallTool(response http.ResponseWriter, request *http.Request) {
	if !s.executionQueryAuthorized(request) {
		writeError(response, http.StatusUnauthorized, "unauthorized", "Invalid internal query auth.", false)
		return
	}
	body, ok := s.readJSONBody(response, request)
	if !ok {
		return
	}
	envelope, err := decodePersonalMemoryRecallToolEnvelope(body)
	if err != nil {
		writeError(response, http.StatusBadRequest, "invalid_request", err.Error(), false)
		return
	}
	result, err := s.personalMemoryRecallTool.PersonalMemoryRecall(
		request.Context(),
		executionapplication.PersonalMemoryRecallRequest{
			AccessContext: agentToolAccessContextToApplication(envelope.AccessContext),
			Query:         envelope.Payload.Query,
		},
	)
	if err != nil {
		if executionapplication.IsAgentToolConflict(err) {
			writeError(response, http.StatusConflict, "run_context_conflict", err.Error(), false)
			return
		}
		writeError(
			response,
			http.StatusServiceUnavailable,
			"personal_memory_recall_failed",
			"Personal memory recall failed.",
			true,
		)
		return
	}
	writeJSON(response, http.StatusOK, result)
}

func decodePersonalMemoryRecallToolEnvelope(body json.RawMessage) (personalMemoryRecallToolEnvelope, error) {
	var envelope personalMemoryRecallToolEnvelope
	decoder := json.NewDecoder(bytes.NewReader(body))
	decoder.DisallowUnknownFields()
	if err := decoder.Decode(&envelope); err != nil {
		return envelope, fmt.Errorf("personal_memory_recall request is invalid: %w", err)
	}
	var trailing any
	if err := decoder.Decode(&trailing); !errors.Is(err, io.EOF) {
		return envelope, errors.New("personal_memory_recall request has trailing JSON content")
	}
	envelope.Payload.Query = strings.TrimSpace(envelope.Payload.Query)
	envelope.AccessContext.RequestID = strings.TrimSpace(envelope.AccessContext.RequestID)
	envelope.AccessContext.InvocationID = strings.TrimSpace(envelope.AccessContext.InvocationID)
	envelope.AccessContext.WorkflowStepID = strings.TrimSpace(envelope.AccessContext.WorkflowStepID)
	if envelope.Payload.Query == "" || len([]rune(envelope.Payload.Query)) > 400 {
		return envelope, errors.New("payload.query must contain between 1 and 400 characters")
	}
	if envelope.AccessContext.RequestID == "" ||
		envelope.AccessContext.InvocationID == "" || envelope.AccessContext.WorkflowStepID == "" ||
		envelope.AccessContext.ActivityAttempt <= 0 {
		return envelope, errors.New("access_context.request_id, invocation_id, workflow_step_id, and activity_attempt are required")
	}
	return envelope, nil
}

func agentToolAccessContextToApplication(
	access agentToolAccessContext,
) executionapplication.AgentToolAccessContext {
	return executionapplication.AgentToolAccessContext{
		ExcludedResultRefs: access.ExcludedResultRefs,
		ActivityAttempt:    access.ActivityAttempt,
		DeadlineAt:         parseDeadlineAt(access.DeadlineAt),
		CitationStart:      access.CitationStart,
		InvocationID:       access.InvocationID,
		RequestID:          access.RequestID,
		ResultLimit:        access.ResultLimit,
		UserGoal:           access.UserGoal,
		WorkflowStepID:     access.WorkflowStepID,
	}
}

func parseDeadlineAt(value string) time.Time {
	if value == "" {
		return time.Time{}
	}
	parsed, err := time.Parse(time.RFC3339Nano, value)
	if err != nil {
		return time.Time{}
	}
	return parsed
}
