import { describe, expect, it } from "vitest";
import { exactSharedSecret, loadAgentRuntimeServerConfig } from "../src/server-config.ts";

describe("agent runtime server config", () => {
	it.each([undefined, ""])("keeps an absent shared secret unconfigured", (value) => {
		expect(exactSharedSecret(value)).toBe("");
	});

	it("preserves the configured shared secret byte-for-byte", () => {
		expect(exactSharedSecret("secret-value")).toBe("secret-value");
	});

	it.each([" secret-value", "secret-value ", "secret-value\n", "\tsecret-value"])(
		"rejects surrounding whitespace in a configured shared secret",
		(value) => {
			expect(() => exactSharedSecret(value)).toThrow(
				"AGENT_CONTROL_RUNTIME_SHARED_SECRET must not contain surrounding whitespace",
			);
		},
	);
});

describe("agent runtime server binding", () => {
	it("uses the production-compatible defaults", () => {
		expect(loadAgentRuntimeServerConfig({})).toEqual({
			host: "0.0.0.0",
			port: 27062,
			metricsPort: 9091,
		});
	});

	it("uses the configured host and port", () => {
		expect(loadAgentRuntimeServerConfig({ HOST: "127.0.0.1", PORT: "27162" })).toEqual({
			host: "127.0.0.1",
			port: 27162,
			metricsPort: 9091,
		});
	});

	it.each(["0", "65536", "not-a-port"])("rejects invalid port %s", (port) => {
		expect(() => loadAgentRuntimeServerConfig({ PORT: port })).toThrow("PORT");
	});
});
