package main

import (
	"context"
	"fmt"
	"github.com/MUC-WIKI/campus-agent/apps/agent-control/internal/config"
	"github.com/MUC-WIKI/campus-agent/apps/agent-control/internal/runtimeassembly"
	"log/slog"
	"net/http"
	"os"
	"os/signal"
	"syscall"
	"time"
)

func main() {
	logger := slog.New(slog.NewJSONHandler(os.Stdout, nil))
	if err := run(logger); err != nil {
		logger.Error("execution_worker.runtime_stopped", "event_type", "execution_worker.runtime_stopped", "status", "failed", "error_code", "runtime_stopped", "source", "process")
		os.Exit(1)
	}
}
func run(logger *slog.Logger) error {
	cfg, err := config.LoadWorkerFromEnv()
	if err != nil {
		return fmt.Errorf("load worker config: %w", err)
	}
	ctx, stop := signal.NotifyContext(context.Background(), syscall.SIGINT, syscall.SIGTERM)
	defer stop()
	owner, err := runtimeassembly.NewWorker(cfg, logger)
	if err != nil {
		return err
	}
	runtime, err := owner.Start(ctx, logger)
	if err != nil {
		owner.Close()
		return err
	}
	server := &http.Server{Addr: fmt.Sprintf("%s:%d", cfg.Host, cfg.Port), Handler: owner.Handler(runtime), ReadHeaderTimeout: 5 * time.Second, ReadTimeout: 5 * time.Second, WriteTimeout: 5 * time.Second, IdleTimeout: 30 * time.Second}
	metricsServer := &http.Server{Addr: fmt.Sprintf("%s:9090", cfg.Host), Handler: runtime.MetricsHandler(), ReadHeaderTimeout: 5 * time.Second, WriteTimeout: 5 * time.Second, IdleTimeout: 30 * time.Second}
	errors := make(chan error, 2)
	go func() {
		if err := metricsServer.ListenAndServe(); err != nil && err != http.ErrServerClosed {
			errors <- err
		}
	}()
	go func() {
		if err := server.ListenAndServe(); err != nil && err != http.ErrServerClosed {
			errors <- err
		}
	}()
	select {
	case <-ctx.Done():
	case err = <-runtime.Errors():
	case err = <-errors:
	}
	runtime.StartDraining()
	done := make(chan struct{})
	go func() {
		runtime.Stop()
		shutdownCtx, cancel := context.WithTimeout(context.Background(), 5*time.Second)
		defer cancel()
		_ = server.Shutdown(shutdownCtx)
		_ = metricsServer.Shutdown(shutdownCtx)
		owner.Close()
		close(done)
	}()
	select {
	case <-done:
		return err
	case <-time.After(365 * time.Second):
		return fmt.Errorf("worker shutdown exceeded 365s")
	}
}
